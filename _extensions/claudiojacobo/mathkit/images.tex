% images.tex — every image in the PDF fits a box as wide as its column and
% three-quarters as tall: the PDF version of the web page's side-by-side tiles
% (_cells.scss gives them all a 4:3 box), so images in a layout row come out
% the same height. A full-width image is at most 0.75 × 6.5in ≈ 4.9in tall.
% Images only shrink, never grow, and explicit sizes are capped too (Quarto
% gives mermaid diagrams their full natural width, often wider than the page).
% Pandoc's own \pandocbounded only fits the column's width and the page's
% height. Change the height limit with
%   \renewcommand*\mk@imgmaxheight{0.9\linewidth}   (inside \makeatletter)
\makeatletter
\newsavebox\mk@imgbox
\newcommand*\mk@imgmaxheight{0.75\linewidth}
\NewCommandCopy\mk@includegraphics\includegraphics
\RenewDocumentCommand\includegraphics{s O{} m}{%
  \sbox\mk@imgbox{\IfBooleanTF{#1}%
    {\mk@includegraphics*[#2]{#3}}{\mk@includegraphics[#2]{#3}}}%
  \Gscale@div\@tempa{\dimexpr\mk@imgmaxheight\relax}%
                    {\dimexpr\ht\mk@imgbox+\dp\mk@imgbox\relax}%
  \Gscale@div\@tempb{\linewidth}{\wd\mk@imgbox}%
  \ifdim\@tempb\p@<\@tempa\p@\let\@tempa\@tempb\fi
  \ifdim\@tempa\p@<\p@\scalebox{\@tempa}{\usebox\mk@imgbox}%
  \else\usebox\mk@imgbox\fi}
\makeatother
